534: static int sm5440_hw_init(struct sm5440_direct *sm)
535: {
536: 	int reg;
537: 	int ret;
538: 	int i;
539: 
540: 	ret = i2c_smbus_write_byte_data(sm->client, SM5440_REG_CNTL1,
541: 					 SM5440_CNTL1_SW_RESET);
542: 	if (ret)
543: 		return ret;
544: 	for (i = 0; i < 255; i++) {
545: 		usleep_range(1000, 2000);
546: 		reg = i2c_smbus_read_byte_data(sm->client, SM5440_REG_CNTL1);
547: 		if (reg < 0)
548: 			return reg;
549: 		if (!(reg & SM5440_CNTL1_SW_RESET))
550: 			break;
551: 	}
552: 	if (i == 255)
553: 		return -ETIMEDOUT;
554: 
555: #define SM5440_WRITE(_reg, _val) do {					\
556: 	ret = i2c_smbus_write_byte_data(sm->client, (_reg), (_val));	\
557: 	if (ret)							\
558: 		return ret;						\
559: } while (0)
560: 
561: 	SM5440_WRITE(SM5440_REG_CNTL1, SM5440_CNTL1_WDT_30S);
562: 	SM5440_WRITE(SM5440_REG_CNTL2, 0xf2);
563: 	SM5440_WRITE(SM5440_REG_CNTL3, 0xb8);
564: 	SM5440_WRITE(SM5440_REG_CNTL4, 0xff);
565: 	SM5440_WRITE(SM5440_REG_CNTL6, 0x09);
566: 	/* CNTL7 (switching frequency) is programmed per charge current in
567: 	 * sm5440_start(), the way the vendor driver does it. */
568: 	SM5440_WRITE(SM5440_REG_VBUSCNTL, 0x07);
569: 	SM5440_WRITE(SM5440_REG_VBATCNTL,
570: 		     ((SM5440_VBATREG_MV - 3800) * 10) / 125);
571: 	SM5440_WRITE(SM5440_REG_VOUTCNTL, 0x3f);
572: 	SM5440_WRITE(SM5440_REG_IBUSCNTL, SM5440_INITIAL_IBUS_MA / 50);
573: 	SM5440_WRITE(SM5440_REG_PRTNCNTL, 0xfe);
574: 	SM5440_WRITE(SM5440_REG_THEMCNTL1, 0x0c);
575: 	SM5440_WRITE(SM5440_REG_ADCCNTL1,
576: 		     SM5440_ADCCNTL1_AVG_32 |
577: 		     SM5440_ADCCNTL1_CONTINUOUS |
578: 		     SM5440_ADCCNTL1_ENABLE);
579: 	SM5440_WRITE(SM5440_REG_ADCCNTL2, 0xdf);
580: #undef SM5440_WRITE
581: 
582: 	/* Reading the four interrupt latches clears stale bootloader events. */
583: 	for (i = 0; i < 4; i++) {
584: 		ret = i2c_smbus_read_byte_data(sm->client, i);
585: 		if (ret < 0)
586: 			return ret;
587: 	}
588: 
589: 	return 0;
590: }
